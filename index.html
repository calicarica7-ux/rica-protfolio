Java Code

Cycling Distance Analyzer

This Java program records the distances of 12 cyclists and calculates the total distance, average distance, farthest distance, and number of cyclists who traveled at least 20 km.

import java.util.Scanner;

public class CyclingDistanceAnalyzer {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);

        double[] distances = new double[12];
        double total = 0;
        double average;
        double farthest;
        int farthestIndex = 0;
        int count = 0;

        // Input distances for 12 cyclists
        for (int i = 0; i < distances.length; i++) {
            while (true) {
                System.out.print("Enter distance for cyclist "
                        + (i + 1) + " (km): ");

                if (input.hasNextDouble()) {
                    double distance = input.nextDouble();

                    if (distance >= 0) {
                        distances[i] = distance;
                        break;
                    } else {
                        System.out.println(
                                "Invalid input! Distance must be nonnegative.");
                    }
                } else {
                    System.out.println(
                            "Invalid input! Please enter a number.");
                    input.next();
                }
            }
        }

        // Calculate total distance
        for (int i = 0; i < distances.length; i++) {
            total += distances[i];

            if (distances[i] >= 20) {
                count++;
            }
        }

        // Calculate average distance
        average = total / distances.length;

        // Find the cyclist who traveled the farthest
        farthest = distances[0];

        for (int i = 1; i < distances.length; i++) {
            if (distances[i] > farthest) {
                farthest = distances[i];
                farthestIndex = i;
            }
        }

        // Display results
        System.out.println("\nCYCLING DISTANCE RESULTS");
        System.out.printf("Total distance: %.2f km%n", total);
        System.out.printf("Average distance: %.2f km%n", average);
        System.out.printf("Farthest distance: %.2f km%n", farthest);
        System.out.println("Cyclist who traveled the farthest: "
                + (farthestIndex + 1));
        System.out.println("Array index of farthest cyclist: "
                + farthestIndex);
        System.out.println("Cyclists who traveled at least 20 km: "
                + count);

        input.close();
    }
}
